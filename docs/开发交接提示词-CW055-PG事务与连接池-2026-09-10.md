  -e POSTGRES_PASSWORD=testpass \
  -e POSTGRES_DB=customer_v3_test \

# 等就绪（约 3-5 秒）
until docker exec cw055-pg-test pg_isready -U testuser -q; do sleep 1; done

### 5.2 L1 机制层：SQLSTATE→幂等恢复矩阵（新增在 `db_pg.py`）

新增模块级常量与纯函数（**不要改 `pg_transaction` 的既有语义**，只在其外层加分类器，避免波及 149 个调用点）：

```python
# 新增（放在 _ALLOWED_ISOLATION_LEVELS 之后）
PG_SQLSTATE_QUERY_CANCELED = "57014"        # statement_timeout / 客户端取消
PG_SQLSTATE_DEADLOCK_DETECTED = "40P01"
PG_SQLSTATE_SERIALIZATION_FAILURE = "40001"
PG_SQLSTATE_UNIQUE_VIOLATION = "23505"
PG_SQLSTATE_LOCK_NOT_AVAILABLE = "55P03"
PG_SQLSTATE_ADMIN_SHUTDOWN = "57P01"        # 断连/服务端关闭
PG_SQLSTATE_CONNECTION_FAILURE = "08006"
PG_SQLSTATE_IN_FAILED_SQL_TRANSACTION = "25P02"

class PgTxnVerdict(Enum):
    """SQLSTATE → 幂等恢复动作的分类结果。"""
    RETRYABLE_IDEMPOTENT = "retryable_idempotent"   # 40001/40P01/55P03：安全重试，须带幂等键
    BOUNDED_FAILURE = "bounded_failure"             # 57014/池耗尽：有界失败，不重试，脱敏上报
    CONNECTION_RECYCLE = "connection_recycle"       # 08006/57P01/25P02：连接不得返池
    BUSINESS_ROLLBACK = "business_rollback"         # 其他：完整回滚，原样上抛

def classify_pg_failure(exc: BaseException) -> PgTxnVerdict: ...
```

要求：
- `classify_pg_failure` 必须**只读 `exc.sqlstate`**（psycopg 的 `Error.sqlstate`），不得靠异常类名字符串匹配（`psycopg.errors.*` 是动态生成的子类，字符串匹配脆弱）；`psycopg_pool.PoolTimeout` 无 sqlstate → 单独 `isinstance` 分支归 `BOUNDED_FAILURE`。
- 池耗尽：`pg_transaction` 外新增 `pg_connection_bounded()` 上下文或在 `get_pg_pool()` 的 `pool.connection(timeout=…)` 处捕获 `PoolTimeout`，**日志必须走 `redact_postgres_dsn()`**，且断言日志文本不含 `testpass`/密码/完整 DSN。
- 异常事务回收：断言「未提交连接不返池」。psycopg_pool 在 `conn.transaction()` 内异常时会自动 rollback 后归还；要补的是**验证**——用一个故意泄漏事务的用例（在 `yield` 后不退出上下文就断连），验证下一次借用拿到的连接 `in_transaction is False` 且能正常提交。
- 断连：`docker restart cw055-pg-test` 或 `SELECT pg_terminate_backend(pid)` 制造，断言 `check=ConnectionPool.check_connection`（L251）把死连接剔除，下一请求成功。
